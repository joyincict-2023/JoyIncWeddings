sidebarTitle: "Package Estimator"
description: "Answer a few quick questions and Joy Inc. will recommend the wedding coordination package that best fits your celebration, with an estimated investment."
---

export const Estimator = () => {
  const PRICES = { essentials: 299, keepsake: 399, gathering: 549, celebration: 749 };
  const NAMES = {
    essentials: "The Elopement Essentials",
    keepsake: "The Intimate Keepsake",
    gathering: "The Joyful Gathering",
    celebration: "The Ultimate Celebration",
  };

  const totalSteps = 5;
  const [step, setStep] = useState(1);
  const [type, setType] = useState(null);
  const [hours, setHours] = useState(null);
  const [rehearsal, setRehearsal] = useState(null);
  const [support, setSupport] = useState(null);
  const [events, setEvents] = useState([]);
  const [error, setError] = useState(false);
  const [done, setDone] = useState(false);

  const toggleEvent = (v) => {
    setEvents((prev) => prev.includes(v) ? prev.filter((x) => x !== v) : [...prev, v]);
  };

  const isValid = () => {
    if (step === 1) return !!type;
    if (step === 2) return !!hours;
    if (step === 3) return !!rehearsal;
    if (step === 4) return !!support;
    if (step === 5) return events.length > 0;
    return true;
  };

  const next = () => {
    if (!isValid()) { setError(true); return; }
    setError(false);
    if (step < totalSteps) setStep(step + 1);
    else setDone(true);
  };

  const back = () => {
    setError(false);
    if (step > 1) setStep(step - 1);
  };

  const restart = () => {
    setStep(1); setType(null); setHours(null); setRehearsal(null);
    setSupport(null); setEvents([]); setError(false); setDone(false);
  };

  const recommend = () => {
    const h = Number(hours);
    let key = "gathering";
    let reason = "Your celebration sounds like a great fit for full wedding-day coordination, with someone there to keep the details moving so you can enjoy the day.";
    if (type === "elopement" || type === "micro") {
      key = "essentials";
      reason = "Your celebration sounds intimate enough that you may not need full-day coverage. The Essentials gives you focused coordination without paying for support you may not need.";
    } else if (type === "intimate" && h <= 6) {
      key = "keepsake";
      reason = "Your smaller celebration would likely benefit from focused coordination before and during the wedding, while keeping the package streamlined.";
    }
    if (h >= 8 || events.length >= 5 || rehearsal === "yes") {
      if (key === "essentials") key = "keepsake";
      else if (key === "keepsake") key = "gathering";
    }
    if (support === "hands" || h > 8 || type === "complex") {
      key = "celebration";
      reason = "Your answers suggest a wedding with enough moving pieces that additional support before the wedding, and a second set of hands on the wedding day, could make a meaningful difference.";
    }
    if (type === "elopement" && support !== "hands" && h <= 4) {
      key = "essentials";
    }
    return { key, reason };
  };

  const optionStyle = {
    display: "flex",
    alignItems: "center",
    gap: "10px",
    border: "1px solid #e8e1cf",
    borderRadius: "14px",
    padding: "15px 16px",
    margin: "10px 0",
    cursor: "pointer",
    fontFamily: 'Georgia, "Times New Roman", serif',
    color: "#2d2d2d",
    background: "#fff",
  };

  const cardStyle = {
    background: "#fff",
    border: "1px solid #e8e1cf",
    borderRadius: "18px",
    padding: "28px",
    boxShadow: "0 8px 30px rgba(0,0,0,.06)",
    fontFamily: 'Georgia, "Times New Roman", serif',
    color: "#2d2d2d",
    maxWidth: "780px",
    margin: "24px auto",
  };

  const btnBase = {
    border: 0,
    borderRadius: "999px",
    padding: "13px 20px",
    fontFamily: 'Georgia, "Times New Roman", serif',
    fontWeight: 600,
    fontSize: "16px",
    cursor: "pointer",
  };

  if (done) {
    const { key, reason } = recommend();
    return (
      <div style={{ background: "#fffaf0", padding: "24px", borderRadius: "18px" }}>
        <div style={cardStyle}>
          <div style={{ textAlign: "center", paddingTop: "8px" }}>
            <div style={{ textTransform: "uppercase", letterSpacing: "2px", fontSize: "12px", color: "#777", fontWeight: "bold" }}>
              Based on your answers, we recommend
            </div>
            <div style={{ fontSize: "34px", lineHeight: 1.15, margin: "8px 0", color: "#2d2d2d" }}>{NAMES[key]}</div>
            <div style={{ fontSize: "42px", fontWeight: "bold", margin: "10px 0", color: "#2d2d2d" }}>
              ${PRICES[key].toLocaleString()} estimated
            </div>
            <p style={{ maxWidth: "580px", margin: "12px auto 20px", color: "#555", lineHeight: 1.6 }}>{reason}</p>
            <p style={{ fontSize: "13px", color: "#777" }}>
              This is an estimated investment, not a final quote. Your exact needs may affect the final price.
            </p>
            <a
              href="mailto:joyincict@gmail.com?subject=Joy%20Inc.%20Wedding%20Coordination%20Consultation"
              style={{ display: "inline-block", textDecoration: "none", color: "#222", background: "#ffde59", padding: "14px 22px", borderRadius: "999px", fontWeight: "bold", marginTop: "8px" }}
            >
              Let's Talk About Your Wedding
            </a>
            <div>
              <button onClick={restart} style={{ ...btnBase, marginTop: "16px", background: "transparent", textDecoration: "underline", color: "#2d2d2d" }}>
                Start Over
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  }

  const options = {
    1: {
      name: "type",
      question: "What kind of celebration are you planning?",
      items: [
        ["elopement", "Elopement"],
        ["micro", "Micro-wedding"],
        ["intimate", "Small or intimate wedding"],
        ["traditional", "Traditional wedding"],
        ["complex", "Large or logistically complex wedding"],
      ],
      value: type, set: setType, kind: "radio",
    },
    2: {
      name: "hours",
      question: "How much wedding-day coverage do you think you'll need?",
      items: [["4", "About 4 hours"], ["6", "About 6 hours"], ["8", "About 8 hours"], ["10", "More than 8 hours"]],
      value: hours, set: setHours, kind: "radio",
    },
    3: {
      name: "rehearsal",
      question: "Will you have a rehearsal?",
      items: [["yes", "Yes, we'll have a rehearsal"], ["no", "No rehearsal"]],
      value: rehearsal, set: setRehearsal, kind: "radio",
    },
    4: {
      name: "support",
      question: "How much support would you like before the wedding?",
      items: [
        ["final", "Mainly final details and wedding-day coordination"],
        ["meeting", "A planning meeting plus final details support"],
        ["hands", "Multiple planning conversations, vendor coordination, and hands-on support"],
      ],
      value: support, set: setSupport, kind: "radio",
    },
    5: {
      name: "event",
      question: "Which parts of your wedding will need coordination?",
      subtext: "Choose all that apply.",
      items: [
        ["firstlook", "First look / pre-ceremony photos"],
        ["ceremony", "Ceremony"],
        ["cocktail", "Cocktail hour"],
        ["reception", "Reception"],
        ["dinner", "Dinner / formalities"],
        ["sendoff", "Send-off"],
      ],
      kind: "checkbox",
    },
  };

  const current = options[step];

  return (
    <div style={{ background: "#fffaf0", padding: "24px", borderRadius: "18px" }}>
      <div style={cardStyle}>
        <h1 style={{ fontSize: "clamp(30px,5vw,44px)", lineHeight: 1.08, margin: "0 0 10px", color: "#2d2d2d" }}>
          Find Your Perfect Level of Support
        </h1>
        <p style={{ color: "#666", marginTop: 0, lineHeight: 1.6 }}>
          Answer a few quick questions and we'll recommend the Joy Inc. coordination package that best fits your celebration.
        </p>

        <div style={{ height: "8px", background: "#eee8d8", borderRadius: "99px", overflow: "hidden", margin: "22px 0 28px" }}>
          <div style={{ height: "100%", width: `${(step / totalSteps) * 100}%`, background: "#ffde59", transition: ".25s" }} />
        </div>

        <div>
          <h2 style={{ fontSize: "24px", margin: "0 0 12px", color: "#2d2d2d" }}>{current.question}</h2>
          {current.subtext && <p style={{ fontSize: "13px", color: "#777", marginTop: 0 }}>{current.subtext}</p>}
          {current.items.map(([v, label]) => (
            <label key={v} style={optionStyle}>
              {current.kind === "checkbox" ? (
                <input
                  type="checkbox"
                  checked={events.includes(v)}
                  onChange={() => toggleEvent(v)}
                  style={{ marginRight: "10px", accentColor: "#222" }}
                />
              ) : (
                <input
                  type="radio"
                  name={current.name}
                  checked={current.value === v}
                  onChange={() => current.set(v)}
                  style={{ marginRight: "10px", accentColor: "#222" }}
                />
              )}
              <span>{label}</span>
            </label>
          ))}
          {error && (
            <div style={{ color: "#a33", fontSize: "14px", marginTop: "10px" }}>
              {step === 5 ? "Please choose at least one event." : "Please choose an option to continue."}
            </div>
          )}
        </div>

        <div style={{ display: "flex", gap: "12px", justifyContent: "space-between", marginTop: "24px" }}>
          <button
            onClick={back}
            disabled={step === 1}
            style={{ ...btnBase, background: "#f0eee7", color: "#333", opacity: step === 1 ? 0.45 : 1, cursor: step === 1 ? "not-allowed" : "pointer" }}
          >
            Back
          </button>
          <button
            onClick={next}
            style={{ ...btnBase, background: "#ffde59", color: "#222" }}
          >
            {step === totalSteps ? "See My Recommendation" : "Next"}
          </button>
        </div>
      </div>
    </div>
  );
};

<Estimator />
